DFD - STRIDE Threat Model Demos

Basic DFD

      dfd-beta
        title Simple Web App
        direction LR

        external user "Web User"

        boundary web_tier "Web Tier" {
          process web "Web Server"
          process api "API Service"
        }

        boundary data_tier "Data Tier" {
          datastore db "User Database"
        }

        user -- "HTTP request" --> web
        web -- "REST call" --> api
        api -- "SQL query" --> db

DFD with Threats

      dfd-beta showThreats
        title Payment System Threat Model

        external customer "Customer"
        external bank "Payment Gateway"

        boundary app "Application" {
          process gateway "API Gateway"
          process auth "Auth Service"
          process payments "Payment Processor"
        }

        boundary storage "Data Layer" {
          datastore orders "Order Database"
          datastore keys "Key Vault"
        }

        f1: customer -- "payment request" --> gateway
        gateway -- "authenticate" --> auth
        auth -- "fetch keys" --> keys
        gateway -- "process payment" --> payments
        payments -- "charge" --> bank
        payments -- "store order" --> orders

        threat customer S "Attacker impersonates customer" severity high status investigate
        threat customer S "Session hijacking via stolen token" severity medium status new
        threat gateway D "DDoS flooding API" severity high status mitigated
        threat auth S "Brute force login" severity critical status new
        threat f1 T "Man-in-the-middle attack" severity high status mitigated
        threat orders I "Unauthorized data access" severity medium status new
        threat payments E "Privilege escalation to admin" severity critical status investigate
