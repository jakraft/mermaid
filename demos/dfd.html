DFD - STRIDE Threat Model Demos

Basic DFD

      dfd-beta
        title Simple Web App
        direction LR

        external user "Web User"

        boundary web_tier "Web Tier" {
          process web "Web Server"
          process api "API Service"
        }

        boundary data_tier "Data Tier" {
          datastore db "User Database"
        }

        user -- "HTTP request" --> web
        web -- "REST call" --> api
        api -- "SQL query" --> db

DFD with Threats

      dfd-beta showThreats
        title Payment System Threat Model

        external customer "Customer"
        external bank "Payment Gateway"

        boundary app "Application" {
          process gateway "API Gateway"
          process auth "Auth Service"
          process payments "Payment Processor"
        }

        boundary storage "Data Layer" {
          datastore orders "Order Database"
          datastore keys "Key Vault"

        f1: customer -- "payment request" --> gateway
        gateway -- "authenticate" --> auth
        auth -- "fetch keys" --> keys
        gateway -- "process payment" --> payments
        payments -- "charge" --> bank
        payments -- "store order" --> orders

        threat customer S "Attacker impersonates customer" severity high status investigate
        threat customer S "Session hijacking via stolen token" severity medium status new
        threat gateway D "DDoS flooding API" severity high status mitigated
        threat auth S "Brute force login" severity critical status new
        threat f1 T "Man-in-the-middle attack" severity high status mitigated
        threat orders I "Unauthorized data access" severity medium status new
        threat payments E "Privilege escalation to admin" severity critical status investigate

Healthcare Portal — Full STRIDE Analysis

      dfd-beta showThreats
        title Healthcare Portal Threat Model
        direction LR

        external patient "Patient Browser"
        external doctor "Doctor Mobile App"
        external lab "Lab System"
        external insurer "Insurance API"

        boundary dmz "DMZ" {
          process lb "Load Balancer"
          process waf "WAF"
        }

        boundary internal "Internal Network" {
          boundary app_tier "Application Tier" {
            process api "API Gateway"
            process auth "Auth Service"
            process records "Records Service"
            process notify "Notification Service"
            process billing "Billing Engine"
          }

          boundary data_tier "Data Tier" {
            datastore ehr "EHR Database"
            datastore audit "Audit Log"
            datastore cache "Session Cache"
            datastore docs "Document Store"
          }
        }

        patient -- "HTTPS request" --> lb
          """
            Patient browser sends HTTPS requests through
            the public internet. All traffic is TLS 1.3
            encrypted with certificate pinning.
          """
        doctor -- "mTLS request" --> lb
          """
            Doctor mobile app uses mutual TLS with
            client certificates for authentication.
          """
        lb -- "forward" --> waf
        waf -- "filtered request" --> api
        api -- "authenticate" --> auth
        auth -- "session check" --> cache
        api -- "patient data" --> records
          """
            Requests include patient ID, record type,
            and date range filters. PHI is transmitted
            in encrypted payload fields.
          """
        records -- "read/write EHR" --> ehr
        records -- "store documents" --> docs
        records -- "audit trail" --> audit
        api -- "send alert" --> notify
        notify -- "push notification" --> doctor
        api -- "submit claim" --> billing
        billing -- "claim request" --> insurer
        billing -- "billing record" --> ehr
        lab -- "lab results" --> api

        threat patient S "Credential stuffing attack" severity high status investigate
        threat patient S "Session token theft via XSS" severity critical status mitigated
        threat lb D "Volumetric DDoS attack" severity high status mitigated
        threat waf T "WAF rule bypass via encoding" severity medium status not-applicable
        threat api E "Broken access control to admin endpoints" severity critical status new
        threat api D "API rate limit exhaustion" severity medium status mitigated
        threat auth S "OAuth token forgery" severity high status investigate
        threat auth R "Failed login attempts not logged" severity medium status new
        threat records I "Patient data exposure via IDOR" severity critical status investigate
        threat records T "Unauthorized record modification" severity high status new
        threat ehr I "Database backup exfiltration" severity high status investigate
        threat ehr T "SQL injection via search" severity critical status mitigated
        threat audit R "Audit log tampering" severity high status new
        threat cache T "Session fixation attack" severity medium status investigate
        threat docs I "Unencrypted PHI at rest" severity critical status new
        threat notify S "Spoofed push notifications" severity medium status not-applicable
        threat billing E "Price manipulation via API" severity high status investigate
        threat billing I "Claim data leak to third party" severity medium status new

Auto-Numbered Flows with Subsequences

      dfd-beta autonumber
        title API Request Lifecycle
        direction LR

        external client "Client App"

        boundary gateway_zone "Gateway" {
          process gw "API Gateway"
          process authz "Auth Service"
        }

        boundary backend "Backend Services" {
          process orders "Order Service"
          process inventory "Inventory Service"
          datastore db "Order Database"
          datastore cache "Redis Cache"
        }

        client -- "API request" --> gw {
          gw -- "validate token" --> authz
          authz -- "check session" --> cache
        }
        gw -- "create order" --> orders {
          orders -- "check stock" --> inventory {
            inventory -- "read inventory" --> cache
            inventory -- "update stock" --> db
          }
          orders -- "persist order" --> db
        }
        orders -- "order confirmation" --> gw
        gw -- "response" --> client

Auto-Numbered with Threats

      dfd-beta showThreats autonumber
        title Secure File Upload
        direction LR

        external user "User"

        boundary app "Application" {
          process upload "Upload Service"
          process scan "Virus Scanner"
          process encrypt "Encryption Service"
        }

        boundary storage "Storage" {
          datastore files "File Store"
          datastore meta "Metadata DB"
        }

        user -- "upload file" --> upload {
          upload -- "scan file" --> scan
          upload -- "encrypt file" --> encrypt {
            encrypt -- "store encrypted" --> files
          }
          upload -- "save metadata" --> meta
        }
        upload -- "upload result" --> user

        threat user S "Impersonate another user" severity high status investigate
        threat upload T "Malicious file upload" severity critical status new
        threat upload D "Large file DoS" severity medium status mitigated
        threat scan T "Scanner bypass via polyglot" severity high status investigate
        threat files I "Unencrypted data at rest" severity critical status mitigated
